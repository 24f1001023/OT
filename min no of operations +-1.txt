def min_ops_to_sort_non_decreasing_dp():
    arr = list(map(int, input("Enter space-separated array elements: ").split()))
    n = len(arr)
    sorted_arr = sorted(arr)
    dp = [0] * n 
    
    for i in range(n):
        prev_dp = dp[:]
        min_prev = prev_dp[0]
        for j in range(n):
            if prev_dp[j] < min_prev:
                min_prev = prev_dp[j]
            dp[j] = min_prev + abs(arr[i] - sorted_arr[j])
            
    print(f"Minimum operations required: {min(dp)}")
min_ops_to_sort_non_decreasing_dp()
